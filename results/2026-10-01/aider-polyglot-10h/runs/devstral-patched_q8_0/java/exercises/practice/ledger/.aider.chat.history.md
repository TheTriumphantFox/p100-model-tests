import java.time.LocalDate;
import java.time.format.DateTimeFormatter;
import java.util.ArrayList;
import java.util.List;

public class Ledger {
    public LedgerEntry createLedgerEntry(String d, String desc, int c) {
        LedgerEntry le = new LedgerEntry();
        le.setChange(c);
        le.setDescription(desc);
        le.setLocalDate(LocalDate.parse(d));
        return le;
    }

    public String format(String cur, String loc, LedgerEntry[] entries) {
        validateCurrency(cur);
        validateLocale(loc);

        FormatConfig config = createFormatConfig(cur, loc);
        String header = config.getHeader();

        if (entries.length == 0) {
            return header;
        }

        List<LedgerEntry> sortedEntries = sortEntries(entries);
        StringBuilder result = new StringBuilder(header);

        for (LedgerEntry entry : sortedEntries) {
            String date = formatDate(entry.getLocalDate(), config.getDatePattern());
            String description = truncateDescription(entry.getDescription());
            String amount = formatAmount(entry.getChange(), config);

            result.append("\n");
            result.append(String.format("%s | %-25s | %13s",
                    date,
                    description,
                    amount));
        }

        return result.toString();
    }

    private void validateCurrency(String cur) {
        if (!cur.equals("USD") && !cur.equals("EUR")) {
            throw new IllegalArgumentException("Invalid currency");
        }
    }

    private void validateLocale(String loc) {
        if (!loc.equals("en-US") && !loc.equals("nl-NL")) {
            throw new IllegalArgumentException("Invalid locale");
        }
    }

    private FormatConfig createFormatConfig(String cur, String loc) {
        String currencySymbol = getCurrencySymbol(cur);
        String datePattern = getDatePattern(loc);
        String decimalSeparator = getDecimalSeparator(loc);
        String thousandSeparator = getThousandSeparator(loc);
        String header = getHeader(loc);

        return new FormatConfig(currencySymbol, datePattern, decimalSeparator,
                              thousandSeparator, header);
    }

    private String getCurrencySymbol(String cur) {
        return cur.equals("USD") ? "$" : "€";
    }

    private String getDatePattern(String loc) {
        return loc.equals("en-US") ? "MM/dd/yyyy" : "dd/MM/yyyy";
    }

    private String getDecimalSeparator(String loc) {
        return loc.equals("en-US") ? "." : ",";
    }

    private String getThousandSeparator(String loc) {
        return loc.equals("en-US") ? "," : ".";
    }

    private String getHeader(String loc) {
        return loc.equals("en-US") ?
            "Date       | Description               | Change       " :
            "Datum      | Omschrijving              | Verandering  ";
    }

    private List<LedgerEntry> sortEntries(LedgerEntry[] entries) {
        List<LedgerEntry> negatives = new ArrayList<>();
        List<LedgerEntry> positives = new ArrayList<>();

        for (LedgerEntry entry : entries) {
            if (entry.getChange() >= 0) {
                positives.add(entry);
            } else {
                negatives.add(entry);
            }
        }

        negatives.sort((e1, e2) -> e1.getLocalDate().compareTo(e2.getLocalDate()));
        positives.sort((e1, e2) -> e1.getLocalDate().compareTo(e2.getLocalDate()));

        List<LedgerEntry> all = new ArrayList<>();
        all.addAll(negatives);
        all.addAll(positives);

        return all;
    }

    private String formatDate(LocalDate date, String pattern) {
        return date.format(DateTimeFormatter.ofPattern(pattern));
    }

    private String truncateDescription(String description) {
        if (description.length() > 25) {
            return description.substring(0, 22) + "...";
        }
        return description;
    }

    private String formatAmount(double change, FormatConfig config) {
        boolean isNegative = change < 0;
        double absoluteValue = Math.abs(change);
        double amountInCurrency = absoluteValue / 100;

        String formatted = String.format("%.02f", amountInCurrency);
        String[] parts = formatted.split("\\.");
        String integerPart = parts[0];
        String decimalPart = parts[1];

        String formattedInteger = formatIntegerPart(integerPart, config.getThousandSeparator());
        String formattedAmount = config.getCurrencySymbol() + formattedInteger +
                               config.getDecimalSeparator() + decimalPart;

        return applyNegativeFormatting(formattedAmount, isNegative, config.getLocale());
    }

    private String formatIntegerPart(String integerPart, String thousandSeparator) {
        StringBuilder result = new StringBuilder();
        int count = 1;

        for (int i = integerPart.length() - 1; i >= 0; i--) {
            if ((count % 3) == 0 && i > 0) {
                result.insert(0, thousandSeparator);
            }
            result.insert(0, integerPart.charAt(i));
            count++;
        }

        return result.toString();
    }

    private String applyNegativeFormatting(String amount, boolean isNegative, String locale) {
        if (!isNegative) {
            return locale.equals("nl-NL") ? " " + amount + " " : amount + " ";
        }

        if (locale.equals("en-US")) {
            return "(" + amount + ")";
        } else {
            return amount.replaceFirst("\\$", "$ -") + " ";
        }
    }

    private static class FormatConfig {
        private final String currencySymbol;
        private final String datePattern;
        private final String decimalSeparator;
        private final String thousandSeparator;
        private final String header;
        private final String locale;

        public FormatConfig(String currencySymbol, String datePattern,
                          String decimalSeparator, String thousandSeparator,
                          String header) {
            this.currencySymbol = currencySymbol;
            this.datePattern = datePattern;
            this.decimalSeparator = decimalSeparator;
            this.thousandSeparator = thousandSeparator;
            this.header = header;
            this.locale = datePattern.equals("MM/dd/yyyy") ? "en-US" : "nl-NL";
        }

        public String getCurrencySymbol() { return currencySymbol; }
        public String getDatePattern() { return datePattern; }
        public String getDecimalSeparator() { return decimalSeparator; }
        public String getThousandSeparator() { return thousandSeparator; }
        public String getHeader() { return header; }
        public String getLocale() { return locale; }
    }

    public static class LedgerEntry {
        LocalDate localDate;
        String description;
        double change;

        public LocalDate getLocalDate() {
            return localDate;
        }

        public void setLocalDate(LocalDate localDate) {
            this.localDate = localDate;
        }

        public String getDescription() {
            return description;
        }

        public void setDescription(String description) {
            this.description = description;
        }

        public double getChange() {
            return change;
        }

        public void setChange(double change) {
            this.change = change;
        }
    }
}
