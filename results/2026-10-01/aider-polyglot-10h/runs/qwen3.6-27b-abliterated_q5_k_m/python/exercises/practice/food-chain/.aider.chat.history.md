To generate the lyrics for "I Know an Old Lady Who Swallowed a Fly", I will implement the `recite` function in `food_chain.py`.

The function needs to:
1.  Define the animals and their specific reaction lines.
2.  Iterate from `start_verse` to `end_verse`.
3.  For each verse, construct the lines:
    *   The opening line: "I know an old lady who swallowed a {animal}."
    *   The specific reaction line for that animal (except for the fly, which has no specific reaction line other than the ending, and the horse, which ends the song).
    *   The cumulative lines: "She swallowed the {current} to catch the {previous}..." with special handling for the spider ("that wriggled and jiggled and tickled inside her").
    *   The ending line: "I don't know why she swallowed the fly. Perhaps she'll die." (except for the horse).
4.  Join the verses with empty strings between them (as per standard song formatting in these exercises, usually a blank line between verses).

def recite(start_verse, end_verse):
    animals = [
        "fly",
        "spider",
        "bird",
        "cat",
        "dog",
        "goat",
        "cow",
        "horse"
    ]

    reactions = [
        "",
        "It wriggled and jiggled and tickled inside her.",
        "How absurd to swallow a bird!",
        "Imagine that, to swallow a cat!",
        "What a hog, to swallow a dog!",
        "Just opened her throat and swallowed a goat!",
        "I don't know how she swallowed a cow!",
        "She's dead, of course!"
    ]

    verses = []
    for i in range(start_verse, end_verse + 1):
        idx = i - 1
        animal = animals[idx]
        verse = [f"I know an old lady who swallowed a {animal}."]
        
        if idx > 0:
            verse.append(reactions[idx])
        
        if idx == 7:
            # Horse verse ends early
            pass
        else:
            # Cumulative part
            for j in range(idx, 0, -1):
                predator = animals[j]
                prey = animals[j-1]
                if prey == "spider":
                    verse.append(f"She swallowed the {predator} to catch the {prey} that wriggled and jiggled and tickled inside her.")
                else:
                    verse.append(f"She swallowed the {predator} to catch the {prey}.")
            
            verse.append("I don't know why she swallowed the fly. Perhaps she'll die.")
        
        verses.append("\n".join(verse))

    return "\n\n".join(verses)
