	// 2. Calculate product.

			}
		}
	}
	
	if len(palindromes) == 0 {
		return Product{}, Product{}, errors.New("no palindromes found")

	minVal := -1
	maxVal := -1
	
	for p := range palindromes {

	}

	return smallest, largest, nil

	s := strconv.Itoa(n)
	len := len(s)
	for i := 0; i < len/2; i++ {
