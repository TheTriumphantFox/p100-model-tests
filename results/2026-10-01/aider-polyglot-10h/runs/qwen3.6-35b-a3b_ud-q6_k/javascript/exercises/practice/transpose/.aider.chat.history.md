The algorithm will:
1. Split the input string into lines.
2. Determine the maximum line length to know how many rows the transposed output will have.
3. Iterate through each column index (which becomes the row index in the output).
4. For each output row, iterate through the original lines (columns).
5. Collect characters, handling padding:
   - If a character exists at the current position in the original line, add it.
   - If not, check if any subsequent original line has a character at this column index. If so, pad with a space (left padding). If not, stop adding to this output row (no right padding).
6. Join the collected characters for each output row and join all output rows with newlines.

export const transpose = (input) => {
  if (!input) {
    return [];
  }

  const lines = input.split('\n');
  const maxLen = Math.max(...lines.map((line) => line.length), 0);

  const result = [];

  for (let i = 0; i < maxLen; i++) {
    let row = '';
    for (let j = 0; j < lines.length; j++) {
      const char = lines[j][i];
      if (char !== undefined) {
        row += char;
      } else {
        // Check if there are any more characters in subsequent lines for this column
        let hasMore = false;
        for (let k = j + 1; k < lines.length; k++) {
          if (lines[k][i] !== undefined) {
            hasMore = true;
            break;
          }
        }
        if (hasMore) {
          row += ' ';
        } else {
          break;
        }
      }
    }
    result.push(row);
  }

  return result;
};

export const transpose = (input) => {
  if (!input || input.length === 0) {
    return [];

  const maxLen = Math.max(...input.map((line) => line.length), 0);
  const result = [];
